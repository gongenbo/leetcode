/**
*
给定数字n，输出符合以下规律的n x n旋转数字矩阵

n=3

6 7 9

2 5 8

1 3 4



n=4

7 13 14 16

6 8 12 15

2 5 9 11

1 3 4 10
*/
public class RotateMatrix {
public static int[][] rotMatrix(int n){

        int[][] matrix = new int[n][n];
        int col = 0;
        int row = n-1;
        int num = 1;
        //int flag = 0;//斜对角，反斜对角
        matrix[row][col] = num;
        num++;
        while(col<=n-1 && row >=0){
            //斜对角
            row--;
            while(col<=n-1 && row>=0 && row<=n-1){
                matrix[row][col] = num;
                col++;
                row++;
                num++;
            }
            //反斜对角
            col++;
            while(col<=n-1 && row>=0 && row<=n-1 && col >=0){
                matrix[row][col] = num;
                col--;
                row--;
                num++;
            }
        }
        return matrix;
    }

    //对角线：正确版本
    public static int[][] rotateMatrix(int n) {
        int[][] mat = new int[n][n];
        int num = 1;

        // d 表示对角线编号
        for (int d = 0; d <= 2 * (n - 1); d++) {

            // 每条对角线的行范围
            int rowStart = Math.max(0, d - (n - 1));
            int rowEnd = Math.min(n - 1, d);

            if (d % 2 == 0) {
                // 偶数对角线：从下往上
                for (int r = rowEnd; r >= rowStart; r--) {
                    int c = d - r;
                    mat[n - 1 - r][c] = num++;
                }
            } else {
                // 奇数对角线：从上往下
                for (int r = rowStart; r <= rowEnd; r++) {
                    int c = d - r;
                    mat[n - 1 - r][c] = num++;
                }
            }
        }
        return mat;
    }
    public static void main(String[] args) {
        // Scanner in = new Scanner(System.in);
        // // 注意 hasNext 和 hasNextLine 的区别
        // while (in.hasNextInt()) { // 注意 while 处理多个 case
        //     int a = in.nextInt();
        //     int b = in.nextInt();
        //     System.out.println(a + b);
        // }
        int n = 5;
        int[][] matrix = rotateMatrix(n);
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                System.out.print(matrix[i][j] + " ");
            }
            System.out.println();
        }
    }
}
